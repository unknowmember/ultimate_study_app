import 'dart:async';
import 'package:flutter/material.dart';
import 'package:webview_windows/webview_windows.dart';

class YtMusicMiniPlayer extends StatefulWidget {
  const YtMusicMiniPlayer({super.key});

  @override
  State<YtMusicMiniPlayer> createState() => _YtMusicMiniPlayerState();
}

class _YtMusicMiniPlayerState extends State<YtMusicMiniPlayer> {
  final _controller = WebviewController();
  bool _isInitialized = false;
  bool _isExpanded = true;
  String? _errorMessage;

  @override
  void initState() {
    super.initState();
    _initWebView();
  }

  Future<void> _initWebView() async {
    setState(() {
      _errorMessage = null;
      _isInitialized = false;
    });

    print('[YT_MUSIC_LOG] Step 1: Bắt đầu gọi _initWebView()');

    try {
      print('[YT_MUSIC_LOG] Step 2: Đang gọi _controller.initialize()...');
      
      await _controller.initialize().timeout(
        const Duration(seconds: 10),
        onTimeout: () {
          throw TimeoutException('Khởi tạo WebView2 Controller bị treo quá 10 giây.');
        },
      );
      
      print('[YT_MUSIC_LOG] Step 3: Khởi tạo WebView2 thành công!');

      print('[YT_MUSIC_LOG] Step 4: Đang cài đặt PopupWindowPolicy...');
      await _controller.setPopupWindowPolicy(WebviewPopupWindowPolicy.deny);

      print('[YT_MUSIC_LOG] Step 5: Đang load URL https://music.youtube.com ...');
      await _controller.loadUrl('https://music.youtube.com');
      
      print('[YT_MUSIC_LOG] Step 6: Load URL hoàn tất!');

      if (mounted) {
        setState(() => _isInitialized = true);
      }
    } catch (e, stack) {
      print('=== [YT_MUSIC ERROR] ===');
      print('Nội dung lỗi: $e');
      print('Stack trace: $stack');
      
      if (mounted) {
        setState(() {
          _errorMessage = 'Lỗi phát sinh từ Code/System:\n$e';
        });
      }
    }
  }

  @override
  void dispose() {
    _controller.dispose();
    super.dispose();
  }

  @override
  Widget build(BuildContext context) {
    return AnimatedContainer(
      duration: const Duration(milliseconds: 300),
      curve: Curves.easeInOut,
      width: _isExpanded ? 420 : 320,
      height: _isExpanded ? 520 : 64,
      decoration: BoxDecoration(
        color: const Color(0xFF1E1E2C),
        borderRadius: BorderRadius.circular(16),
        border: Border.all(color: Colors.redAccent.withOpacity(0.4), width: 1.5),
        boxShadow: [
          BoxShadow(
            color: Colors.black.withOpacity(0.5),
            blurRadius: 16,
            offset: const Offset(0, 6),
          ),
        ],
      ),
      child: Column(
        children: [
          // Header
          InkWell(
            onTap: () => setState(() => _isExpanded = !_isExpanded),
            borderRadius: BorderRadius.circular(16),
            child: Container(
              height: 60,
              padding: const EdgeInsets.symmetric(horizontal: 16),
              child: Row(
                children: [
                  const Icon(Icons.music_note, color: Colors.redAccent, size: 24),
                  const SizedBox(width: 12),
                  const Expanded(
                    child: Text(
                      'YouTube Music',
                      style: TextStyle(fontWeight: FontWeight.bold, fontSize: 14, color: Colors.white),
                    ),
                  ),
                  Icon(
                    _isExpanded ? Icons.keyboard_arrow_down : Icons.keyboard_arrow_up,
                    color: Colors.white70,
                  ),
                ],
              ),
            ),
          ),

          // Content
          if (_isExpanded)
            Expanded(
              child: _errorMessage != null
                  ? Padding(
                      padding: const EdgeInsets.all(16.0),
                      child: Column(
                        mainAxisAlignment: MainAxisAlignment.center,
                        children: [
                          const Icon(Icons.error_outline, color: Colors.redAccent, size: 48),
                          const SizedBox(height: 12),
                          Text(
                            _errorMessage!,
                            textAlign: TextAlign.center,
                            style: const TextStyle(color: Colors.white70, fontSize: 13),
                          ),
                          const SizedBox(height: 16),
                          ElevatedButton.icon(
                            style: ElevatedButton.styleFrom(
                              backgroundColor: Colors.redAccent,
                              foregroundColor: Colors.white,
                            ),
                            onPressed: _initWebView,
                            icon: const Icon(Icons.refresh, size: 18),
                            label: const Text('Thử lại'),
                          ),
                        ],
                      ),
                    )
                  : _isInitialized
                      ? ClipRRect(
                          borderRadius: const BorderRadius.vertical(bottom: Radius.circular(16)),
                          child: Webview(_controller),
                        )
                      : const Center(
                          child: CircularProgressIndicator(color: Colors.redAccent),
                        ),
            ),
        ],
      ),
    );
  }
}
